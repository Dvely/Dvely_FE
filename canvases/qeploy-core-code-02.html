export function useAgentRun(projectId: string) {
    const queryClient = useQueryClient();

    const sendMessage = useMutation({
        mutationFn: (content: string) =>
            agentApi.createRun({ projectId, content }),
        onSuccess: (run) => {
            setPreviewUrl(run.previewUrl);
            setApproval(run.approval);
            appendChat(run.summary);
            queryClient.invalidateQueries({ queryKey: ['project', projectId] });
        },
        onError: (err) => toast.error(err.message),
    });

    const approve = useMutation({
        mutationFn: (runId: string) => agentApi.approve(runId),
        onSuccess: () => {
            queryClient.invalidateQueries({
                queryKey: ['agent-runs', projectId],
            });
        },
    });

    return { sendMessage, approve };
}
