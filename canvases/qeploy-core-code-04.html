@Transactional
public AgentRunResult handle(AgentCommand command) {
    Project project = projectRepository.getById(command.projectId());
    Intent intent = decisionAgent.classify(command.message());

    AgentRun run = agentRunRepository.save(
        AgentRun.started(project.getId(), command.actorId(),
            intent.type()));

    AgentPlan plan = agentRouter.route(project, intent);
    if (plan.requiresApproval()) {
        approvalService.request(run.getId(), plan.toApprovalPayload());
        run.markWaitingApproval();
    } else {
        agentExecutor.execute(run.getId(), plan);
    }

    auditLogService.record(command.actorId(), project.getId(),
        AuditAction.AGENT_PLAN, "AgentRun", run.getId());
    return AgentRunResult.from(run, plan);
}
