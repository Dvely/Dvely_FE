@Transactional
public Deployment deploy(Project project, DeployCommand command) {
    deploymentPolicy.validateOwner(project, command.actorId());
    deploymentPolicy.validateBranch(project, command.branch());

    WorkflowRun workflow = githubActions.dispatch(
        project.repoUrl(), command.branch(), Map.of(
            "version", command.version(),
            "environment", command.environment()
        ));

    Deployment deployment = Deployment.queued(
        project.getId(), command.version(), workflow.id(),
        workflow.logUrl());
    deploymentRepository.save(deployment);

    auditLogService.record(command.actorId(), project.getId(),
        AuditAction.DEPLOY_REQUEST, "Deployment", deployment.getId());
    return deployment;
}
